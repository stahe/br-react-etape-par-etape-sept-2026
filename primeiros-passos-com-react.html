    // =========================================================================
// 01-first-application/src/App.tsx - um primeiro componente
// -------------------------------------------------------------------------
// Um componente React é uma FUNÇÃO, cujo nome começa com letra
// maiúscula, que RETORNA o que deve ser exibido, escrito em JSX: HTML
// dentro do código TypeScript. O React chama essa função para montar a
// página; ele a chama DE NOVO a cada vez que o estado do componente muda, e
// atualiza a página com o novo resultado.
// =========================================================================

import { useState } from 'react';
// uma folha de estilo, importada pelo componente que a utiliza
import './App.css';

// dados "comuns", fora do componente: eles nunca mudam
const title = 'Welcome to React';
const languages = ['TypeScript', 'HTML', 'CSS'];

export default function App() {
  // useState(0): um pedaço de ESTADO, com valor inicial 0. Obtemos seu
  // valor ATUAL (counter) e a função que o altera (setCounter).
  // Chamar setCounter pede ao React que chame App DE NOVO para atualizar
  // a página. Nunca modificamos a página nós mesmos.
  const [counter, setCounter] = useState(0);
  const [time, setTime] = useState(() => new Date().toLocaleTimeString());

  // uma função chamada pelo botão
  function increment() {
    setCounter((n) => n + 1);
    setTime(new Date().toLocaleTimeString());
  }

  // o JSX retornado: { expressão } insere o valor de uma expressão;
  // className substitui o atributo HTML class (uma palavra reservada em JavaScript)
  return (
    <div className="container py-4">
      <h1 className="h3">{title}</h1>
      <p>This page is built in the browser, by React, from index.html, main.tsx and the App component.</p>
      <p>
        It is written in {languages.join(', ')} ({languages.length} languages).
      </p>

      {/* onClick: a função a chamar ao clicar (cf. exemplo 04) */}
      <button className="btn btn-primary" onClick={increment}>
        Click me
      </button>
      <p className="mt-3 counter">
        You clicked {counter} times (last click: {time}).
      </p>
    </div>
  );
}
    • Linha 11: o hook [useState], fornecido pelo [React];
    • linha 13: a folha de estilos do componente;
    • linhas 16–17: dois dados comuns, declarados fora do componente: eles nunca mudam;
    • linha 19: o componente: uma função cujo nome começa com uma letra maiúscula (é assim que o JSX distingue um componente, [<App />], de um elemento HTML, [<div>]). [export default]: é o que o módulo exporta por padrão, importado em [main.tsx] por [import App from './App'];
    • Linha 24: [useState(0)] cria uma variável de estado com valor inicial 0. Ele retorna um array de dois elementos, que “desempacotamos”: o valor atual ([counter]) e a função que o altera ([setCounter]). O estado é armazenado pelo [React]: ele persiste de uma chamada da função para a seguinte;
    • Linha 25: Você pode passar ao [useState] uma função que calcula o valor inicial; ela é chamada apenas na primeira vez que a página é exibida;
    • Linhas 28–30: Uma função chamada pelo botão. Ela não modifica a página; ela altera o estado. [setCounter((n) => n + 1)] calcula o novo valor a partir do antigo. Cada chamada à função [setXxx] instrui o [React] a chamar [App]; o [React] compara o novo resultado com o antigo e atualiza apenas as partes da página que mudaram;
    • linhas 35–51: o JSX retornado, ou seja, o que deve ser exibido:
      • o JSX se parece com o HTML, com algumas diferenças: o atributo [class] é escrito como [className] ([class] é uma palavra reservada do [JavaScript]), cada tag deve ser fechada ([<br />], [<input />]) e toda a estrutura deve ter UM único elemento raiz;
      • Linha 37: [{ expression }] insere o valor de uma expressão [TypeScript]. Ele é escapado: se [title] contivesse [<b>], [<b>] seria exibido sem que a tag fosse interpretada;
      • linha 40: a expressão pode ser qualquer coisa: uma chamada de método, um cálculo;
      • linha 43: um comentário em JSX é escrito como [{/* ... */}];
      • linha 44: clicar no botão chama a função [increment] (veja o Exemplo 04). A própria função é fornecida sem parênteses: [onClick={increment()}] a chamaria imediatamente, durante a exibição;
      • linha 48: o valor atual do estado. Quando o estado muda, [App] é chamado novamente, e este texto muda.

    O JSX não é o mesmo que HTML: é uma notação conveniente para chamadas de função. O compilador transforma [<h1 className="h3">{title}</h1>] em uma chamada da forma [jsx('h1', { className: 'h3', children: title })], que cria um objeto descrevendo o elemento. O componente retorna, portanto, uma árvore de objetos (o “DOM virtual”), que o [React] compara com a árvore anterior para determinar o que alterar na página real.

    // =========================================================================
// 02-state/src/App.tsx - estado, valores calculados, useEffect
// -------------------------------------------------------------------------
// Uma pequena calculadora de preços com IVA incluso:
//   - useState    : um valor de ESTADO; alterá-lo (setXxx) chama o
//                   componente de novo, que monta a página com o novo valor;
//   - um valor CALCULADO a partir do estado NÃO é um estado: é uma
//     simples variável, recalculada a cada chamada do componente;
//   - useMemo     : a mesma coisa, mas o cálculo só é refeito se suas
//                   dependências mudaram (útil para um cálculo custoso);
//   - useEffect   : código executado DEPOIS da exibição, quando alguns
//                   valores mudaram: um "efeito colateral" (agir sobre algo
//                   diferente da página: aqui, o título da aba).
// =========================================================================

import { useEffect, useMemo, useState } from 'react';

// um formatador de valores em euros, à maneira inglesa (€1,234.50)
const euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });

export default function App() {
  const [label, setLabel] = useState('Pen');
  const [priceExclVat, setPriceExclVat] = useState(2.5);
  const [quantity, setQuantity] = useState(1);
  const [vatRate, setVatRate] = useState(20); // em %

  // Valores CALCULADOS: simples constantes, recalculadas a cada
  // chamada do componente. Estão sempre atualizadas, nada a sincronizar
  const totalExclVat = quantity * priceExclVat;
  const vatAmount = (totalExclVat * vatRate) / 100;
  // useMemo(cálculo, [dependências]): mesmo resultado, mas o React só refaz o
  // cálculo se alguma das dependências mudou desde a chamada anterior
  const totalInclVat = useMemo(() => totalExclVat + vatAmount, [totalExclVat, vatAmount]);

  // o histórico dos totais com IVA: depende dos valores SUCESSIVOS do
  // total. O total antigo é mantido no estado; se ele difere do
  // novo, o estado é ajustado DURANTE a chamada do componente (o React
  // então reinicia a chamada imediatamente, antes de exibir qualquer coisa)
  const [history, setHistory] = useState<{ id: number; text: string }[]>([]);
  const [oldTotal, setOldTotal] = useState(totalInclVat);
  if (totalInclVat !== oldTotal) {
    setOldTotal(totalInclVat);
    const text = `${euros.format(oldTotal)} -> ${euros.format(totalInclVat)}`;
    // cada linha tem um número único (id), que a identifica (cf. key, exemplo 03)
    setHistory((h) => [{ id: (h[0]?.id ?? 0) + 1, text }, ...h].slice(0, 5));
  }

  // useEffect(efeito, [dependências]): o efeito é executado DEPOIS da primeira
  // exibição, e depois a cada exibição em que totalInclVat mudou

  label: string;
  done: boolean;
}

    setLog((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);

  // impede o navegador de enviar o formulário (e recarregar a página)

    event.preventDefault();
    const input = field.current!;
    const label = input.value.trim();
    if (label === '') {
      return;

    // um estado que contém um array: ele é SUBSTITUÍDO por um novo array
    // (o antigo nunca é modificado: o React não perceberia a mudança)

// 05-controlled-fields/src/App.tsx - todos os tipos de campo, "controlados"
// -------------------------------------------------------------------------
// Um campo CONTROLADO exibe um valor do estado (value={...}) e notifica
// o componente de cada mudança (onChange={...}), que atualiza o
// estado. O estado é a única fonte de verdade: digitar no campo modifica
// o estado, e modificar o estado (por código) modifica o campo.
// É o binding "bidirecional", escrito à mão. O que se lê
// no evento depende do campo:
//   - text input, textarea           -> e.target.value (string)
//   - input type="number"            -> e.target.valueAsNumber (NaN se vazio)
//   - checkbox                       -> e.target.checked (booleano)
//   - radio buttons                  -> e.target.value do botão escolhido
//   - select                         -> e.target.value
//   - select multiple                -> e.target.selectedOptions (vários)
//   - input type="date"              -> e.target.value, "yyyy-MM-dd"
// Mesmo formulário do exemplo 18 do curso de NestJS; mas aqui, sem
// servidor: tudo acontece no navegador.
// =========================================================================

import { useState, type FormEvent } from 'react';

const TITLES = ['Mr', 'Mme', 'Melle'];
const HOBBIES = ['sport', 'music', 'reading', 'cinema'];
const COUNTRIES = [
  { code: 'fr', label: 'France' },
  { code: 'be', label: 'Belgium' },
  { code: 'ch', label: 'Switzerland' },
];

export default function App() {
  // um estado por campo do formulário
  const [title, setTitle] = useState('Mme');
  const [name, setName] = useState('');
  const [age, setAge] = useState<number | null>(30);
  const [presentation, setPresentation] = useState('');
  const [newsletter, setNewsletter] = useState(false);
  const [hobbies, setHobbies] = useState<string[]>(['reading']);
  const [country, setCountry] = useState('fr');
  const [languages, setLanguages] = useState<string[]>([]);
  const [birthDate, setBirthDate] = useState('');
  const [sent, setSent] = useState<string | null>(null);

  // todos os valores, reunidos em um objeto (o que seria enviado a um servidor)
  const model = { title, name: name.trim(), age, presentation, newsletter, hobbies, country, languages, birthDate };

  // várias caixas para UM array: o valor é adicionado ou removido
  function toggleHobby(hobby: string, checked: boolean) {
    setHobbies((l) => (checked ? [...l, hobby] : l.filter((x) => x !== hobby)));
  }

  function send(event: FormEvent<HTMLFormElement>) {
    event.preventDefault(); // sem envio ao servidor, sem recarregamento
    setSent(JSON.stringify(model));
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 05 - controlled fields</h1>
      <div className="row g-4">
        <form className="col-md-7" onSubmit={send}>
          <div className="mb-3">
            <span className="form-label d-block">Title</span>
            {/* botões de opção: o mesmo estado; checked indica qual está marcado */}
            {TITLES.map((t) => (
              <div key={t} className="form-check form-check-inline">
                <input type="radio" className="form-check-input" name="title" id={`title-${t}`} value={t}
                       checked={title === t} onChange={(e) => setTitle(e.target.value)} />
                <label className="form-check-label" htmlFor={`title-${t}`}>{t}</label>
              </div>
            ))}
          </div>
          <div className="row g-3 mb-3">
            <div className="col-8">
              <label className="form-label" htmlFor="name">Name</label>
              <input id="name" className="form-control" value={name} onChange={(e) => setName(e.target.value)} />
            </div>
            <div className="col-4">
              <label className="form-label" htmlFor="age">Age</label>
              {/* um campo vazio dá NaN: null é guardado no estado, '' no campo */}
              <input id="age" type="number" className="form-control" value={age ?? ''}
                     onChange={(e) => setAge(Number.isNaN(e.target.valueAsNumber) ? null : e.target.valueAsNumber)} />
            </div>
          </div>
          <div className="mb-3">
            <label className="form-label" htmlFor="presentation">About you (updated when leaving the field)</label>
            {/* um campo NÃO controlado: defaultValue é seu valor inicial, depois
                o navegador o gerencia sozinho; onBlur só notifica ao sair */}

                      onBlur={(e) => setPresentation(e.target.value)} />
          </div>
          <div className="mb-3 form-check">
            {/* uma única caixa: true / false */}
            <input id="newsletter" type="checkbox" className="form-check-input" checked={newsletter}
                   onChange={(e) => setNewsletter(e.target.checked)} />
            <label htmlFor="newsletter" className="form-check-label">Receive the newsletter</label>
          </div>
          <div className="mb-3">
            <span className="form-label d-block">Hobbies</span>
            {HOBBIES.map((l) => (
              <div key={l} className="form-check form-check-inline">
                <input type="checkbox" className="form-check-input" id={`hobby-${l}`} checked={hobbies.includes(l)}
                       onChange={(e) => toggleHobby(l, e.target.checked)} />
                <label className="form-check-label" htmlFor={`hobby-${l}`}>{l}</label>
              </div>
            ))}
          </div>
          <div className="row g-3 mb-3">
            <div className="col-6">
              <label className="form-label" htmlFor="country">Country</label>
              {/* value no <select> (e não selected nas <option>) */}
              <select id="country" className="form-select" value={country} onChange={(e) => setCountry(e.target.value)}>

              </select>
            </div>
            <div className="col-6">
              <label className="form-label" htmlFor="languages">Spoken languages</label>
              {/* multiple: value é um ARRAY; selectedOptions, as opções escolhidas */}
              <select id="languages" multiple size={3} className="form-select" value={languages}
                      onChange={(e) => setLanguages([...e.target.selectedOptions].map((o) => o.value))}>
                <option value="fr">French</option>
                <option value="en">English</option>
                <option value="es">Spanish</option>
              </select>
            </div>
          </div>
          <div className="mb-3">
            <label className="form-label" htmlFor="birthDate">Date of birth</label>
            <input id="birthDate" type="date" className="form-control w-auto" value={birthDate}
                   onChange={(e) => setBirthDate(e.target.value)} />
          </div>
          <button type="submit" className="btn btn-primary">Send</button>
        </form>

        <div className="col-md-5">
          <h2 className="h6">The model, live</h2>
          {/* JSON.stringify(object, null, 2): o JSON, indentado com 2 espaços */}
          <pre className="bg-light p-2 rounded small">{JSON.stringify(model, null, 2)}</pre>
          {sent !== null && (
            <>
              <h2 className="h6">Last sending</h2>
              <pre className="bg-success-subtle p-2 rounded small text-wrap">{sent}</pre>
            </>
          )}
        </div>
      </div>
    </div>
  );
}

// 06-use-reducer/src/App.tsx - um formulário, um objeto, um reducer
// -------------------------------------------------------------------------
// O exemplo 05 tinha um useState por campo. Para um formulário real,
// preferimos UM objeto que contém todos os valores, o MODELO, e
// descrevemos suas mudanças em uma única função, o REDUCER:
//
//     reducer(estadoAtual, action) -> novoEstado
//
// useReducer(reducer, initialState) retorna o estado e uma função
// dispatch(action): o componente não modifica mais o estado, ele DESCREVE o que
// aconteceu ({ type: 'edit', field: 'lastName', value: 'Dupont' }) e o
// reducer deduz dela o novo estado. Toda a lógica fica em um só lugar,
// e é uma função PURA (mesma entrada, mesma saída, sem efeito
// colateral), fácil de testar.
//
// O estado também guarda, para cada campo, se ele foi "tocado" (deixado ao
// menos uma vez); a partir disso deduzimos se ele está "modificado" (seu valor difere do
// valor inicial) e se está desabilitado.
// =========================================================================

import { useReducer, type ChangeEvent } from 'react';

// o formato dos dados do formulário
interface Profile {
  title: string;
  firstName: string;
  lastName: string;
  age: number;
  newsletter: boolean;
  email: string;
  country: string;
  birthDate: string;
  presentation: string;
}

const INITIAL: Profile = {
  title: 'Mme',
  firstName: 'Marie',
  lastName: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  country: 'fr',
  birthDate: '',
  presentation: '',
};

type Field = keyof Profile; // 'title' | 'firstName' | ...: o nome de um campo

interface State {
  values: Profile;
  touched: Partial<Record<Field, boolean>>; // os campos já deixados
}

// as ações possíveis: uma "união discriminada" pela propriedade type
type Action =
  | { type: 'edit'; field: Field; value: Profile[Field] }
  | { type: 'leave'; field: Field }
  | { type: 'uppercase' }
  | { type: 'reset' };

// o reducer: ele NUNCA modifica o estado recebido, retorna um novo
function reducer(state: State, action: Action): State {
  switch (action.type) {
    case 'edit':
      return { ...state, values: { ...state.values, [action.field]: action.value } };
    case 'leave':
      return { ...state, touched: { ...state.touched, [action.field]: true } };
    case 'uppercase': {
      const { firstName, lastName } = state.values;
      return { ...state, values: { ...state.values, firstName: firstName.toUpperCase(), lastName: lastName.toUpperCase() } };
    }
    case 'reset':
      return { values: { ...INITIAL }, touched: {} };
  }
}

const TITLES = ['Mr', 'Mme', 'Melle'];
const TRACKED_FIELDS: Field[] = ['firstName', 'lastName', 'age', 'email'];

export default function App() {
  const [state, dispatch] = useReducer(reducer, { values: { ...INITIAL }, touched: {} });
  const v = state.values;

  // valores DEDUZIDOS do estado
  const modified = (c: Field) => v[c] !== INITIAL[c];
  const disabled = (c: Field) => c === 'email' && !v.newsletter; // o e-mail exige a newsletter
  const formModified = (Object.keys(INITIAL) as Field[]).some(modified);

  // os atributos comuns a um campo de texto: uma função que os monta
  // (id, value, onChange, onBlur, disabled), para "espalhar" no elemento: {...bind('lastName')}
  function bind(field: 'firstName' | 'lastName' | 'email' | 'birthDate' | 'presentation' | 'country') {
    return {
      id: field,
      value: v[field],
      disabled: disabled(field),
      onChange: (e: ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) =>
        dispatch({ type: 'edit', field, value: e.target.value }),
      onBlur: () => dispatch({ type: 'leave', field }),
    };
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 06 - a form, an object, a reducer</h1>
      <div className="row g-4">
        <form className="col-md-7" onSubmit={(e) => e.preventDefault()}>
          <div className="mb-3">
            <span className="form-label d-block">Title</span>
            {TITLES.map((t) => (
              <div key={t} className="form-check form-check-inline">
                <input type="radio" className="form-check-input" id={`title-${t}`} name="title" value={t}
                       checked={v.title === t} onChange={() => dispatch({ type: 'edit', field: 'title', value: t })} />
                <label className="form-check-label" htmlFor={`title-${t}`}>{t}</label>
              </div>
            ))}
          </div>
          <div className="row g-3 mb-3">
            <div className="col-5">
              <label className="form-label" htmlFor="firstName">First name</label>
              {/* {...object}: cada propriedade do objeto se torna um atributo */}
              <input className="form-control" {...bind('firstName')} />
            </div>
            <div className="col-5">
              <label className="form-label" htmlFor="lastName">Last name</label>
              <input className="form-control" {...bind('lastName')} />
            </div>
            <div className="col-2">
              <label className="form-label" htmlFor="age">Age</label>
              <input id="age" type="number" className="form-control" value={v.age}
                     onChange={(e) => dispatch({ type: 'edit', field: 'age', value: e.target.valueAsNumber || 0 })}
                     onBlur={() => dispatch({ type: 'leave', field: 'age' })} />
            </div>
          </div>
          <div className="row g-3 mb-3 align-items-end">
            <div className="col-5">
              <div className="form-check">
                <input id="newsletter" type="checkbox" className="form-check-input" checked={v.newsletter}
                       onChange={(e) => dispatch({ type: 'edit', field: 'newsletter', value: e.target.checked })} />
                <label htmlFor="newsletter" className="form-check-label">Receive the newsletter</label>
              </div>
            </div>
            <div className="col-7">
              <label className="form-label" htmlFor="email">E-mail (if newsletter)</label>
              <input type="email" className="form-control" {...bind('email')} />
            </div>
          </div>
          <div className="row g-3 mb-3">
            <div className="col-6">
              <label className="form-label" htmlFor="country">Country</label>
              <select className="form-select" {...bind('country')}>
                <option value="fr">France</option>
                <option value="be">Belgium</option>
                <option value="ch">Switzerland</option>
              </select>
            </div>
            <div className="col-6">
              <label className="form-label" htmlFor="birthDate">Date of birth</label>
              <input type="date" className="form-control" {...bind('birthDate')} />
            </div>
          </div>
          <div className="mb-3">
            <label className="form-label" htmlFor="presentation">About you</label>
            <textarea rows={2} className="form-control" {...bind('presentation')} />
          </div>
          <div className="d-flex gap-2">
            <button type="button" className="btn btn-outline-primary" onClick={() => dispatch({ type: 'uppercase' })}>
              Name in uppercase
            </button>
            <button type="button" className="btn btn-outline-secondary" onClick={() => dispatch({ type: 'reset' })}>
              Reset
            </button>
          </div>
        </form>

        <div className="col-md-5">
          <h2 className="h6">The model, live</h2>
          <pre className="bg-light p-2 rounded small">{JSON.stringify(v, null, 2)}</pre>
          <h2 className="h6">The state of the fields</h2>
          <table className="table table-sm small">
            <thead><tr><th>field</th><th>touched</th><th>modified</th><th>disabled</th></tr></thead>
            <tbody>
              {TRACKED_FIELDS.map((c) => (
                <tr key={c}>
                  <td>{c}</td>
                  <td>{String(state.touched[c] ?? false)}</td>
                  <td>{String(modified(c))}</td>
                  <td>{String(disabled(c))}</td>
                </tr>
              ))}
            </tbody>
          </table>
          <p className="small mb-0">form modified: <strong>{String(formModified)}</strong></p>
        </div>
      </div>
    </div>
  );
}
    • linhas 25–35: a estrutura de dados do formulário; [INITIAL] (linhas 37–47), seus valores iniciais;

    • Linhas 51–54: o estado completo: os valores e os campos que já foram “visitados” (abandonados ao menos uma vez). [Partial<Record<Field, boolean>>]: um objeto cujas chaves são nomes de campos, todas opcionais;
    • linhas 57–61: as ações possíveis, na forma de uma “união discriminada”: cada ação tem uma propriedade [type], que especifica qual é a ação, e seus próprios dados;
    • linhas 64–77: o reducer: uma função que recebe o estado atual e uma ação e retorna o NOVO estado. Ele nunca modifica o estado recebido (cria uma cópia modificada, [{ ...state, ... }]) e não faz nada além disso: é uma função pura, fácil de entender e de testar. [[action.field]: action.value]: uma propriedade cujo nome é calculado;
    • linha 83: [useReducer(reducer, initialState)] retorna o estado e uma função [dispatch(action)]: o componente não modifica mais o estado; ele descreve o que aconteceu ([dispatch({ type: 'leave', field: 'lastName' })]), e o [React] chama o reducer para deduzir o novo estado;

import { useState, type FormEvent } from 'react';

interface Account {
  title: string;
  firstName: string;
  lastName: string;
  login: string;
  password: string;
  confirmation: string;
}

type Field = keyof Account;
// no máximo uma mensagem de erro por campo
type Errors = Partial<Record<Field, string>>;

const TITLES = ['Mr', 'Mme', 'Melle'];
const LOGIN_FORMAT = /^[a-z0-9._-]{1,30}$/;
const PASSWORD_MIN = 6;

// as regras: uma função pura, que só depende de seus argumentos
function validate(c: Account): Errors {
  const errors: Errors = {};
  if (!TITLES.includes(c.title)) errors.title = 'Choose a title from the list.';
  if (c.firstName.trim() === '') errors.firstName = 'The first name is required.';
  else if (c.firstName.length > 20) errors.firstName = 'The first name must not exceed 20 characters.';
  if (c.lastName.trim() === '') errors.lastName = 'The last name is required.';
  else if (c.lastName.length > 20) errors.lastName = 'The last name must not exceed 20 characters.';
  if (c.login === '') errors.login = 'The login is required.';
  else if (!LOGIN_FORMAT.test(c.login)) errors.login = 'Unaccented lower-case letters, digits, dot, hyphen, underscore.';
  if (c.password === '') errors.password = 'The password is required.';
  else if (c.password.length < PASSWORD_MIN) errors.password = `At least ${PASSWORD_MIN} characters.`;
  // uma regra que compara DOIS campos
  if (c.confirmation !== c.password) errors.confirmation = 'The confirmation does not match the password.';
  return errors;
}

const EMPTY: Account = { title: 'Mr', firstName: '', lastName: '', login: '', password: '', confirmation: '' };
const ALL: Record<Field, boolean> = { title: true, firstName: true, lastName: true, login: true, password: true, confirmation: true };

export default function App() {
  const [account, setAccount] = useState<Account>(EMPTY);
  const [touched, setTouched] = useState<Partial<Record<Field, boolean>>>({});
  const [submitAttempted, setSubmitAttempted] = useState(false);
  const [success, setSuccess] = useState(false);

  // os erros: recalculados a cada exibição, nunca guardados no estado

  // a classe Bootstrap de um campo: vermelho (is-invalid) ou verde (is-valid),
  // somente depois de ele ter sido tocado
  function cssClass(field: Field, base = 'form-control'): string {
    if (!touched[field]) return base;
    return `${base} ${errors[field] ? 'is-invalid' : 'is-valid'}`;
  }

      value: account[field],

      },

  }

          <div className="invalid-feedback">{errors.title}</div>
        </div>
        <div className="row g-3 mb-3">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="firstName">First name</label>
            {/* is-invalid emoldura o campo em vermelho E mostra o
                .invalid-feedback que o segue: o erro do campo é escrito ali */}
            <input className={cssClass('firstName')} {...bind('firstName')} />
            <div className="invalid-feedback">{errors.firstName}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="lastName">Last name</label>
            <input className={cssClass('lastName')} {...bind('lastName')} />
            <div className="invalid-feedback">{errors.lastName}</div>
          </div>
        </div>
        <div className="mb-3">
          <label className="form-label" htmlFor="login">Login</label>
          <input className={cssClass('login')} autoComplete="username" {...bind('login')} />
          <div className="invalid-feedback">{errors.login}</div>
        </div>
        <div className="row g-3 mb-4">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="password">Password</label>
            <input type="password" className={cssClass('password')} autoComplete="new-password" {...bind('password')} />
            <div className="invalid-feedback">{errors.password}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="confirmation">Confirmation</label>
            <input type="password" className={cssClass('confirmation')} autoComplete="new-password" {...bind('confirmation')} />
            <div className="invalid-feedback">{errors.confirmation}</div>
          </div>
        </div>

    </div>
  );

export interface Slot {

  startMinute: number;
  endHour: number;
  endMinute: number;
}

// "8:05" em vez de "8:5"
function time(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

export function timeRange(c: Slot, part: 'all' | 'start' | 'end' = 'all'): string {
  switch (part) {
    case 'start':
      return time(c.startHour, c.startMinute);
    case 'end':
      return time(c.endHour, c.endMinute);
    default:
      return `${time(c.startHour, c.startMinute)} - ${time(c.endHour, c.endMinute)}`;
  }
}

    .split(/\s+/)
    .filter((word) => word !== '')
    .map((word) => word[0].toUpperCase())
    .join('');
}

import { useState } from 'react';
import { timeRange, initials, capitalize, toDate, type Slot } from './formatting';

// datas: dateStyle / timeStyle, ou o detalhe das partes a exibir
const shortDate = new Intl.DateTimeFormat('en-US', { dateStyle: 'medium' });
const fullDate = new Intl.DateTimeFormat('en-US', { dateStyle: 'full' });
const timeOnly = new Intl.DateTimeFormat('en-US', { timeStyle: 'medium' });
const longDay = new Intl.DateTimeFormat('en-US', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric', timeZone: 'UTC' });
const frenchDay = new Intl.DateTimeFormat('fr-FR', { dateStyle: 'long', timeZone: 'UTC' });
// números e valores monetários
const euros = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'EUR' });
const integer = new Intl.NumberFormat('en-US', { maximumFractionDigits: 0 });
const percent = new Intl.NumberFormat('en-US', { style: 'percent', minimumFractionDigits: 1 });

const day = '2026-10-05'; // uma data no formato da API RdvMedecins
const price = 1234.5;
const rate = 0.055;
const name = 'marie pelissier';
const doctor = { title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER' };
const languages = ['TypeScript', 'HTML', 'CSS', 'JavaScript', 'SQL'];
const slots: Slot[] = [
  { startHour: 8, startMinute: 0, endHour: 8, endMinute: 20 },
  { startHour: 8, startMinute: 20, endHour: 8, endMinute: 40 },
  { startHour: 14, startMinute: 5, endHour: 14, endMinute: 25 },
];

export default function App() {
  const [now, setNow] = useState(() => new Date());

  // [código exibido, valor]: uma linha da tabela
  const rows: [string, string][] = [
    ['shortDate.format(now)', shortDate.format(now)],
    ['fullDate.format(now)', fullDate.format(now)],
    ['timeOnly.format(now)', timeOnly.format(now)],
    ['longDay.format(toDate(day))', longDay.format(toDate(day))],
    ['frenchDay.format(toDate(day))', frenchDay.format(toDate(day))],
    ['now.toLocaleDateString()', now.toLocaleDateString()],
    ['euros.format(price)', euros.format(price)],
    ['integer.format(price)', integer.format(price)],
    ['percent.format(rate)', percent.format(rate)],
    ['name.toUpperCase()', name.toUpperCase()],
    ['capitalize(name)', capitalize(name)],
    ["'RDV'.toLowerCase()", 'RDV'.toLowerCase()],
    ["languages.slice(0, 3).join(', ')", languages.slice(0, 3).join(', ')],
    ['JSON.stringify(doctor)', JSON.stringify(doctor)],
    // duas funções em sequência: o resultado da primeira vai para a segunda
    ['initials(capitalize(name))', initials(capitalize(name))],
  ];

  return (
    <div className="container py-4" style={{ maxWidth: 820 }}>
      <h1 className="h3 mb-3">Example 09 - the formatting of values</h1>
      <table className="table table-sm align-middle">
        <thead><tr><th>In the code</th><th>Displayed</th></tr></thead>
        <tbody>
          {rows.map(([code, value]) => (
            <tr key={code}><td><code>{`{${code}}`}</code></td><td>{value}</td></tr>
          ))}
        </tbody>
      </table>

      <h2 className="h6">Object.entries: going through the properties of an object</h2>
      <ul>
        {Object.entries(doctor).map(([key, value]) => (
          <li key={key}>{key} = {value}</li>
        ))}
      </ul>

      <h2 className="h6">A home-made function: timeRange</h2>
      <ul>
        {slots.map((c) => (
          <li key={timeRange(c)}>{timeRange(c)} (start: {timeRange(c, 'start')})</li>
        ))}
      </ul>
      <button className="btn btn-sm btn-outline-primary" onClick={() => setNow(new Date())}>Refresh the time</button>
    </div>
  );
}
    • linhas 13–17: formatadores de data, [Intl.DateTimeFormat(language, options)]: [dateStyle] e [timeStyle] oferecem formatos predefinidos ([full], [long], [medium], [short]); também é possível especificar quais partes exibir ([weekday], [day], [month], [year]). [timeZone: 'UTC']: a data [2026-10-05T00:00:00Z] deve continuar sendo 5 de outubro, qualquer que seja o fuso horário do navegador;
    • Linhas 19–21: formatadores de número: um valor em euros, um número sem casas decimais, uma porcentagem;
    • linhas 39–56: linhas da tabela: o código e seu valor. A última encadeia duas funções;
    • linha 65: o código entre chaves;
    • linhas 72–74: [Object.entries(object)] retorna os pares [chave, valor] de um objeto, que são percorridos com [map];
    • linhas 79–81: a função personalizada [timeRange].

    Os geradores [Intl] são criados de uma vez por todas, fora do componente: criá-los custa mais do que usá-los. O idioma é fixado ali no código ([en-US], e [fr-FR] para a linha em francês); o Exemplo 21 mostrará como seguir o idioma escolhido pelo usuário.
