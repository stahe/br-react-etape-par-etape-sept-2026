// 10-components/src/DoctorCard.tsx - um componente FILHO
// -------------------------------------------------------------------------
// Ele recebe seus dados do pai por meio de PROPS (propriedades): os
// atributos escritos pelo pai em <DoctorCard ... />, reunidos pelo
// React em UM objeto, primeiro parâmetro da função. Seu tipo é
// verificado em tempo de compilação: <DoctorCard doctor={42} /> seria rejeitado.
//
// Regra de ouro: um componente NUNCA MODIFICA suas props. Os dados
// fluem do pai para o filho; para pedir uma mudança,
// o filho chama uma função que o pai lhe passou (cf. exemplo 11).
// =========================================================================

import type { Doctor } from './doctor';
// styles: um objeto { initials: '_initials_x7f3a_1' } - o nome real de cada classe
import styles from './DoctorCard.module.css';

// o tipo das props: o que o pai PODE (?) ou DEVE fornecer
interface Props {
  doctor: Doctor;
  rank: number;
  compact?: boolean; // opcional
}

// o objeto props é "desestruturado"; compact = false: seu valor padrão
export default function DoctorCard({ doctor, rank, compact = false }: Props) {
  // um valor calculado a partir das props: recalculado se o pai as mudar
  const initials = `${doctor.firstName[0]}${doctor.lastName[0]}`;

  return (
    <div className={`card h-100 ${doctor.available ? 'border-success' : ''}`}>
      <div className="card-body d-flex gap-3 align-items-center">
        <div className={styles.initials}>{initials}</div>
        <div>
          <h2 className="h6 mb-1">{rank}. {doctor.title} {doctor.firstName} {doctor.lastName}</h2>
          {!compact && <p className="mb-0 text-muted small">{doctor.specialty}</p>}
        </div>
      </div>
    </div>
  );
}
    • linha 14: uma importação somente de tipo;
    • linha 16: uma importação do módulo CSS (veja abaixo);
    • linhas 19–23: os tipos das props do componente — os atributos que o pai escreve em [<DoctorCard ... />]. [compact?] é opcional;
    • linha 26: o [React] combina as props em UM objeto, que passa como primeiro parâmetro da função. Ele é imediatamente “desempacotado”: [{ doctor, rank, compact = false }] gera três variáveis, com [compact] igual a [false] se o pai não o fornecer. O tipo das props é verificado em tempo de compilação: [<DoctorCard doctor={42} />] seria rejeitado, assim como a omissão de [rank];
    • linha 28: um valor calculado a partir das props: se o pai mudar as props, o componente é reexecutado e o valor é recalculado;
    • linha 33: a classe CSS, lida do módulo CSS;
    • linha 36: a especialidade é exibida apenas no modo normal.

    Regra de ouro: um componente nunca modifica suas próprias props. Os dados fluem do pai para o filho; para solicitar uma mudança, o filho chama uma função que o pai lhe passou (veja o Exemplo 11).

    // =========================================================================
// 10-components/src/App.tsx - o componente PAI
// -------------------------------------------------------------------------
// Ele importa o componente DoctorCard e o usa como uma nova
// tag, <DoctorCard />, quantas vezes for preciso: um componente é
// um pedaço REUTILIZÁVEL de interface, com seus próprios dados, seu JSX e
// seu estilo. Uma tag que começa com letra MAIÚSCULA é um componente;
// em minúscula (<div>, <p>), é um elemento HTML.
// =========================================================================

import { useState } from 'react';
import DoctorCard from './DoctorCard';
import type { Doctor } from './doctor';

export default function App() {
  const [doctors, setDoctors] = useState<Doctor[]>([
    { id: 1, title: 'Mme', firstName: 'Marie', lastName: 'PELISSIER', specialty: 'general practitioner', available: true },
    { id: 2, title: 'Mr', firstName: 'Paul', lastName: 'TATOU', specialty: 'pediatrician', available: false },
    { id: 3, title: 'Melle', firstName: 'Sophie', lastName: 'DUPONT', specialty: 'general practitioner', available: true },
  ]);
  const [compact, setCompact] = useState(false);

  // o pai modifica SEUS dados: o React chama o pai de novo, depois os
  // filhos, com suas novas props
  function toggleAvailability() {
    setDoctors((list) => list.map((m) => ({ ...m, available: !m.available })));
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Example 10 - components and their props</h1>
      <div className="d-flex gap-3 mb-3">
        <label className="form-check-label">
          <input type="checkbox" className="form-check-input" checked={compact}
                 onChange={(e) => setCompact(e.target.checked)} /> compact display
        </label>
        <button className="btn btn-sm btn-outline-primary" onClick={toggleAvailability}>Toggle availabilities</button>
      </div>
      <div className="row g-3">
        {doctors.map((m, i) => (
          <div key={m.id} className="col-md-4">
            {/* doctor={m} passa o OBJETO m (uma expressão);
                doctor="m", sem chaves, passaria a STRING "m" */}
            <DoctorCard doctor={m} rank={i + 1} compact={compact} />
          </div>
        ))}
      </div>
      <p className="mt-3 mb-1 small text-muted">The same card, in compact version, by an attribute without a value:</p>
      <div className="w-50">
        {/* compact, sem valor, é true */}
        <DoctorCard doctor={doctors[0]} rank={1} compact />
      </div>
    </div>
  );
}
    • linha 12: o componente filho é importado como qualquer outro módulo;
    • linhas 25–27: o pai modifica SEUS dados. O [React] o reexecuta e depois reexecuta os filhos com suas novas props: os cartões são atualizados;
    • linha 44: o componente é usado como uma nova tag. [doctor={m}] passa o OBJETO [m] (uma expressão); [doctor="m"], sem chaves, passaria a STRING “m”;
    • linha 51: um atributo sem valor ([compact]) é igual a [true].

    // =========================================================================
// 11-lifting-state-up/src/Counter.tsx
// -------------------------------------------------------------------------
// Um filho que NOTIFICA seu pai: o pai lhe passa FUNÇÕES
// nas props (por convenção, seu nome começa com "on"), e o filho
// as chama quando algo acontece:
//     <Counter label="Coffees" max={3} onIncrement={...} onLimit={...} />
// =========================================================================

import { useState } from 'react';

interface Props {
  label: string;
  max: number;
  onIncrement: (value: number) => void; // com o novo valor
  onLimit: () => void; // sem dados
}

export default function Counter({ label, max, onIncrement, onLimit }: Props) {
  // o estado PRÓPRIO deste contador: cada <Counter> tem o seu
  const [value, setValue] = useState(0);

  function plus() {
    if (value >= max) {
      onLimit();
      return;
    }
    setValue(value + 1);
    // value ainda não foi modificado (será na próxima chamada do
    // componente): value + 1 é passado
    onIncrement(value + 1);
  }

  return (
    <div className="d-flex align-items-center gap-2 mb-2">
      <span className="w-25">{label}</span>
      <button className="btn btn-sm btn-outline-primary" onClick={plus}>+1</button>
      <span className="badge text-bg-secondary">{value} / {max}</span>
    </div>
  );
}
    • Linhas 12–17: duas props são funções: [onIncrement] recebe o novo valor, enquanto [onLimit] não recebe nada. Por convenção, seus nomes começam com “on”, como [onClick];
    • Linha 21: o estado PRÓPRIO deste contador: cada [<Counter>] da página tem o seu. O estado está associado à posição do componente na árvore da página;
    • linhas 23–32: o pai é notificado. Atenção: [setValue(value + 1)] não altera a variável [value] (uma constante fixada para ESTA chamada do componente): ela será [value + 1] na PRÓXIMA chamada. Por isso, [value + 1] é passado ao pai.

// 11-lifting-state-up/src/TextField.tsx - um componente CONTROLADO
// -------------------------------------------------------------------------
// Um campo de formulário "caseiro" (rótulo + campo + ajuda + contador de
// caracteres) usado como um <input> controlado (cf. exemplo 05):
//     <TextField value={lastName} onChange={setLastName} label="Last name" max={20} />
// O componente NÃO tem estado: o valor pertence ao pai, que o passa
// (value) e que ele notifica a cada tecla digitada (onChange). É o
// pai que decide o novo valor.
// =========================================================================

import { useId } from 'react';

interface Props {
  value: string;
  onChange: (value: string) => void;
  label: string;
  max: number;
  help?: string;
}

export default function TextField({ value, onChange, label, max, help = '' }: Props) {
  // useId: um identificador ÚNICO, diferente para cada <TextField> da
  // página, para ligar <label htmlFor> e <input id>
  const id = useId();

  return (
    <div className="mb-3">
      <label className="form-label" htmlFor={id}>{label}</label>
      <input id={id} className={`form-control ${value.length === max ? 'is-invalid' : ''}`} value={value}
             maxLength={max} onChange={(e) => onChange(e.target.value)} />
      <div className="form-text">{help} ({value.length} / {max})</div>
    </div>
  );
}

    Um campo de formulário “personalizado” (rótulo, campo, dica, contador de caracteres) usado como um [<input>] controlado (veja o Exemplo 05): o pai lhe passa o valor ([value]) e uma função chamada a cada tecla digitada ([onChange]).

    • linhas 14–20: as props; [help] é opcional;
    • linha 25: [useId()] fornece um identificador único, diferente para cada [<TextField>] da página, para vincular [<label htmlFor>] e [<input id>];
    • linha 31: cada tecla digitada chama a função do pai com o novo texto.

    O componente não tem estado: o valor pertence ao pai, que determina o novo valor. É o que se chama de “elevar o estado” (lifting state up): os dados compartilhados por vários componentes são armazenados no ancestral comum mais próximo.

// 11-lifting-state-up/src/App.tsx - o pai escuta seus filhos
// -------------------------------------------------------------------------
// Os dados fluem para baixo (props); a informação sobe por meio de
// funções passadas nas props:
//   1. "eventos" do filho: onIncrement, onLimit;
//   2. um componente controlado: value + onChange, como um <input>. O estado
//      continua sendo propriedade do pai, que pode modificá-lo ele mesmo (Clear).
// =========================================================================

import { useState } from 'react';
import Counter from './Counter';
import TextField from './TextField';

export default function App() {
  const [total, setTotal] = useState(0);
  const [message, setMessage] = useState('');
  const [firstName, setFirstName] = useState('Marie');
  const [lastName, setLastName] = useState('PELISSIER');

  // a função chamada por um contador: ela recebe o valor que ele passa
  function onIncrement(label: string, value: number) {
    setTotal((t) => t + 1);
    setMessage(`${label} is now ${value}`);
  }

  function clear() {
    setFirstName('');
    setLastName('');
  }

  return (
    <div className="container py-4" style={{ maxWidth: 640 }}>
      <h1 className="h3 mb-3">Example 11 - the child notifies its parent</h1>

      <div className="card p-3 mb-4">
        <h2 className="h6">1. Functions called by the children</h2>
        <Counter label="Coffees" max={3} onIncrement={(v) => onIncrement('Coffees', v)}
                 onLimit={() => setMessage('No more than 3 coffees!')} />
        <Counter label="Glasses of water" max={8} onIncrement={(v) => onIncrement('Glasses of water', v)}
                 onLimit={() => setMessage('Enough drinking.')} />
        <p className="mb-0">Total of the clicks: {total} - <em>{message}</em></p>
      </div>

      <div className="card p-3">
        <h2 className="h6">2. Controlled components</h2>
        {/* onChange={setFirstName}: a função do estado, passada como está */}
        <TextField value={firstName} onChange={setFirstName} label="First name" max={20} help="Required" />
        <TextField value={lastName} onChange={setLastName} label="Last name" max={20} help="Displayed in upper case" />
        <p className="mb-0">Data of the parent: <strong>{firstName} {lastName.toUpperCase()}</strong></p>
        <button className="btn btn-sm btn-outline-secondary mt-2 align-self-start" onClick={clear}>Clear (from the parent)</button>
      </div>
    </div>
  );
}

    • linha 48: [onChange={setFirstName}]: a função de estado é passada como está; o filho a chama com o novo texto;

// 12-composition/src/Card.tsx - um TEMPLATE reutilizável
// -------------------------------------------------------------------------
// As props passam DADOS... mas também CONTEÚDO: o JSX é
// um valor como outro qualquer (do tipo ReactNode), que pode ser passado a um
// componente e que ele coloca onde quiser:
//   - children: a prop especial que recebe o que o pai escreve ENTRE
//                <Card> e </Card> - o conteúdo "principal";
//   - header, footer: outros espaços reservados, fornecidos pelo pai na
//                forma de props; header tem um conteúdo padrão.
// O card define a moldura (borda, cabeçalho, rodapé), o pai fornece o que
// está dentro. É o equivalente, no lado do cliente, do template layout.hbs
// e de seu {{{body}}} no curso de NestJS.
// =========================================================================

import type { ReactNode } from 'react';

interface Props {
  color?: 'primary' | 'success' | 'danger';
  header?: ReactNode;
  footer?: ReactNode;
  children: ReactNode;
}

export default function Card({ color, header = 'Untitled', footer, children }: Props) {
  return (
    <div className={`card mb-3 ${color ? `border-${color}` : ''}`}>
      <div className={`card-header ${color ? `text-bg-${color}` : ''}`}>{header}</div>
      <div className="card-body">{children}</div>
      {/* nenhum footer fornecido: nenhum rodapé */}
      {footer && <div className="card-footer text-muted small">{footer}</div>}
    </div>
  );
}
    • linhas 18–23: [header], [footer] e [children] são [ReactNode]: JSX, texto, um número ou nada;
    • linha 25: conteúdo padrão;
    • linha 29: [children] é uma propriedade especial: ela recebe tudo o que o pai escreve ENTRE [<Card>] e [</Card>] — o conteúdo “principal”;
    • linha 31: nenhum rodapé fornecido — nenhum rodapé é exibido.

    O template define a moldura (borda, cabeçalho, rodapé); o pai fornece o conteúdo dentro dela: isso equivale à projeção de conteúdo do [Angular] ([<ng-content>]) e aos “slots” do [Vue.js]e, no lado do servidor, ao template [layout.hbs] e seu [{{{body}}}] no curso [NestJS].

    // =========================================================================
// 12-composition/src/DataTable.tsx - uma tabela GENÉRICA
// -------------------------------------------------------------------------
// DataTable<T> é um componente genérico: ele exibe linhas de
// qualquer tipo T. Sabe percorrer as linhas, mas não como
// exibir UMA linha: é o pai que lhe diz, passando-lhe uma
// FUNÇÃO, renderRow(row, index), que retorna o JSX das células.
// É o que se chama de "render prop": o conteúdo fornecido pelo
// pai usa dados que pertencem ao filho (a linha, sua
// posição).
// =========================================================================

import type { ReactNode } from 'react';

interface Props<T> {
  rows: T[];
  columns: string[];
  renderRow: (row: T, index: number) => ReactNode;
  empty?: ReactNode; // exibido quando não há nenhuma linha
}

// <T extends { id: number }>: cada linha deve ter um id (a key)
export default function DataTable<T extends { id: number }>({ rows, columns, renderRow, empty = 'No data.' }: Props<T>) {
  return (
    <table className="table table-sm align-middle mb-0">
      <thead>
        <tr>
          {columns.map((c) => <th key={c}>{c}</th>)}
        </tr>
      </thead>
      <tbody>
        {rows.length === 0 ? (
          <tr>
            <td className="text-muted" colSpan={columns.length}>{empty}</td>
          </tr>
        ) : (
          rows.map((l, i) => <tr key={l.id}>{renderRow(l, i)}</tr>)
        )}
      </tbody>
    </table>
  );
}
    • linhas 15–20: um tipo genérico: as props de um array de linhas do tipo [T], sejam elas quais forem;

// 13-lifecycle/src/Clock.tsx - um componente que inicia e para
// -------------------------------------------------------------------------
// A vida de um componente:
//   1. ele é EXIBIDO (montado): o React chama a função e coloca seu
//      resultado na página; depois executa seus efeitos (useEffect);
//   2. ele é ATUALIZADO, quantas vezes for preciso (seu estado ou suas
//      props mudam);
//   3. ele é REMOVIDO da página (desmontado): o React executa a LIMPEZA de
//      seus efeitos, a função que cada efeito retornou.
// Um relógio inicia um timer (setInterval) quando aparece... e
// deve obrigatoriamente pará-lo quando desaparece: caso contrário o timer
// continuaria rodando à toa (vazamento de memória).
// =========================================================================

import { useEffect, useEffectEvent, useState } from 'react';

interface Props {
  onLog: (message: string) => void;
}

export default function Clock({ onLog }: Props) {
  const [time, setTime] = useState(() => new Date().toLocaleTimeString());

  // useEffectEvent: uma função chamada POR um efeito, que sempre lê
  // as props mais recentes, sem ser uma dependência do efeito. Sem
  // isso, onLog teria de ser colocada nas dependências [...] abaixo,
  // e, como o pai passa uma função NOVA a cada exibição, o
  // timer seria parado e reiniciado a cada segundo
  const log = useEffectEvent((message: string) => onLog(message));

  // []: nenhuma dependência. O efeito roda UMA vez, após a primeira
  // exibição; a função que ele retorna, UMA vez, quando o componente é removido
  useEffect(() => {
    log('Clock: effect - starting the timer');
    const timer = setInterval(() => setTime(new Date().toLocaleTimeString()), 1000);
    // a LIMPEZA
    return () => {
      clearInterval(timer);
      log('Clock: cleanup - timer stopped');
    };
  }, []);

  return <div className="display-6 font-monospace">{time}</div>;
}

    O ciclo de vida de um componente (linhas 4–10): ele é montado (exibido pela primeira vez), atualizado quantas vezes for necessário e, depois, desmontado (removido da página). O [React] não tem métodos separados para cada etapa: tudo passa por [useEffect].

    • Linhas 34–42: [useEffect(effect, [])]: com um array de dependências VAZIO, o efeito é executado uma única vez, após a primeira exibição. A função que ele retorna (linhas 38–41) cuida da limpeza: o [React] a executa quando o componente é removido da página (e antes de cada nova execução do efeito, quando suas dependências mudam). Um relógio inicia um timer ([setInterval]) na inicialização... e deve pará-lo ([clearInterval]) quando desaparece: caso contrário, o timer continuaria rodando à toa e tentaria modificar o estado de um componente que não existe mais;
    • Linha 30: o efeito chama a função [onLog] do pai. A rigor, ele deveria incluí-la em suas dependências: como o efeito usa uma prop, ele precisa ser reativado se a prop mudar. Porém, o pai cria uma função NOVA a cada exibição: o timer seria parado e reiniciado continuamente. [useEffectEvent] ([React] 19.2) cria uma função que sempre chama a versão mais recente de [onLog], sem ser uma dependência do efeito. Ela só pode ser chamada de dentro de um efeito.

// 14-contexts/src/contexts/log.tsx - um contexto para TODA a aplicação
// -------------------------------------------------------------------------
// Como compartilhar dados e funções entre componentes distantes,
// sem passá-los de props em props por todos os
// intermediários? Com um CONTEXTO:
//   1. createContext(defaultValue) cria o contexto;
//   2. um componente PROVIDER, <LogContext value={...}>, dá seu
//      valor a TODOS os seus descendentes, em qualquer profundidade;
//   3. um descendente o LÊ com use(LogContext) (ou useContext).
// Quando o valor fornecido muda, todos os componentes que o leem são
// chamados de novo.
//
// O costume: um arquivo por contexto, que exporta o provider (um
// componente que guarda o estado) e um HOOK para lê-lo (useLog).
// É assim que o React compartilha o que o Angular guarda em seus serviços.
// =========================================================================

import { createContext, use, useCallback, useMemo, useState, type ReactNode } from 'react';

interface Log {
  lines: string[];
  trace: (message: string) => void;
}

// null: o valor lido se NENHUM provider envolver o componente
const LogContext = createContext<Log | null>(null);

// o provider: ele guarda o estado e o fornece a seus descendentes
export function LogProvider({ children }: { children: ReactNode }) {
  const [lines, setLines] = useState<string[]>([]);

  // useCallback: a MESMA função de uma exibição para a seguinte (e não
  // uma nova a cada vez), para que os componentes que a recebem
  // não pareçam ter mudado
  const trace = useCallback((message: string) => {
    setLines((l) => [`${new Date().toLocaleTimeString()} ${message}`, ...l].slice(0, 6));
  }, []);

  // useMemo: um novo objeto somente se lines mudou
  const value = useMemo(() => ({ lines, trace }), [lines, trace]);

  // React 19: o próprio contexto é o componente provider
  return <LogContext value={value}>{children}</LogContext>;
}

// o hook de leitura: um nome que começa com "use"
export function useLog(): Log {
  // use(context): o valor fornecido pelo provider MAIS PRÓXIMO
  const log = use(LogContext);
  if (log === null) {
    throw new Error('useLog must be used inside a <LogProvider>');
  }
  return log;
}

    • linha 27: [createContext(defaultValue)] cria o contexto. [null]: o valor lido por um componente que não está envolvido por nenhum provider;
    • linhas 30–45: o provider: um componente que contém o estado e o repassa a todos os seus descendentes, em qualquer profundidade:
      • linha 36: [useCallback(fn, [dependencies])] passa a MESMA função de uma exibição para a seguinte (enquanto as dependências não mudarem), em vez de criar uma nova. Aqui, não há dependências: [setLines] nunca muda;
      • linha 41: o objeto fornecido só é recriado se [lines] tiver mudado. Sem essas precauções, o provider forneceria um novo objeto a cada exibição, e todos os componentes que leem o contexto seriam chamados desnecessariamente;
      • linha 44: a partir do [React] 19, o próprio contexto é o componente provider (antes, escrevia-se [<LogContext.Provider>]);

  price: number;
}

  total: number;
  add: (item: Item) => void;
  remove: (index: number) => void;
}

const CartContext = createContext<Cart | null>(null);

export function CartProvider({ children }: { children: ReactNode }) {
  const { trace } = useLog();
  const [items, setItems] = useState<Item[]>([]);

  const value = useMemo<Cart>(
    () => ({
      items,
      total: items.reduce((sum, a) => sum + a.price, 0),
      add: (item) => {
        setItems((a) => [...a, item]);
        trace(`cart: + ${item.label}`);

        setItems((a) => a.filter((_, i) => i !== index));

  );

  if (cart === null) {

// -------------------------------------------------------------------------
// Um HOOK PERSONALIZADO é uma função cujo nome começa com "use", e
// que ela mesma chama hooks (useState, useEffect, useRef...). Ele
// permite reutilizar um COMPORTAMENTO (com seu estado) em vários

// -------------------------------------------------------------------------
//     const longPress = useLongPress(() => remove(m), 1000);
//     <button {...longPress.handlers} className={longPress.pressing ? 'press' : ''}>
// A ação só é executada se o botão for mantido pressionado por um

import { useEffect, useRef, useState } from 'react';

export function useLongPress(action: () => void, duration = 1000) {
  const [pressing, setPressing] = useState(false);
  // um valor que sobrevive às exibições SEM provocá-las: useRef
  const timer = useRef<number | undefined>(undefined);

  // quando o componente é removido: nenhum timer órfão
  useEffect(() => () => clearTimeout(timer.current), []);

  function start() {
    setPressing(true);
    timer.current = window.setTimeout(() => {
      setPressing(false);
      action();
    }, duration);
  }

  function stop() {
    clearTimeout(timer.current);
    setPressing(false);
  }

  return {
    pressing,
    handlers: { onPointerDown: start, onPointerUp: stop, onPointerLeave: stop },
  };
}

// 16-confirmation/src/confirmation/confirmation.tsx
// -------------------------------------------------------------------------
// "Tem certeza?": window.confirm() existe, mas não pode nem ser
// estilizado, nem ter seus botões traduzidos. Ele é substituído por um
// diálogo do Bootstrap, controlado por um contexto (cf. exemplo 14).
//
// O truque: confirm(message) retorna uma PROMISE, resolvida com true (Sim)
// ou false (Não) quando o usuário responde. O código chamador é então escrito
// tão simplesmente quanto com window.confirm:
//
//     const confirm = useConfirmation();
//     if (await confirm('Delete this doctor?')) { ... }
//
// O provider exibe o próprio diálogo: ele só precisa ser colocado
// uma vez, em volta de toda a aplicação (cf. main.tsx).
// =========================================================================

import { createContext, use, useCallback, useRef, useState, type ReactNode } from 'react';
import ConfirmDialog from './ConfirmDialog';

type Confirm = (message: string) => Promise<boolean>;

const ConfirmationContext = createContext<Confirm | null>(null);

export function ConfirmationProvider({ children }: { children: ReactNode }) {
  const [message, setMessage] = useState<string | null>(null); // null: diálogo fechado
  // a função que resolverá a promise atual: ela não é exibida,
  // é portanto uma referência e não um estado
  const answer = useRef<((yes: boolean) => void) | null>(null);

  const confirm = useCallback<Confirm>((text) => {
    answer.current?.(false); // uma pergunta ainda em aberto? isso significa "não"
    setMessage(text);
    return new Promise<boolean>((resolve) => {
      answer.current = resolve;
    });
  }, []);

  function respond(yes: boolean) {
    setMessage(null);
    const r = answer.current;
    answer.current = null;
    r?.(yes);
  }

  return (
    <ConfirmationContext value={confirm}>
      {children}
      {message !== null && <ConfirmDialog message={message} onAnswer={respond} />}
    </ConfirmationContext>
  );
}

export function useConfirmation(): Confirm {
  const confirm = use(ConfirmationContext);
  if (confirm === null) {
    throw new Error('useConfirmation must be used inside a <ConfirmationProvider>');
  }
  return confirm;
}

    Dica: [confirm(message)] retorna uma promise, que é resolvida com [true] (Sim) ou [false] (Não) quando o usuário responde. O código chamador fica então tão simples quanto com [window.confirm]: [if (await confirm('Delete this doctor?')) {...}].

    • Linha 22: o que o contexto fornece: uma função;
    • linha 27: a mensagem exibida; [null]: a janela está fechada;
    • linha 30: a função que resolverá a promise atual. Ela não é exibida: é uma referência, não um estado;
    • Linhas 32–38: abrem a janela e retornam uma promise cuja função é mantida em [resolve]. Uma pergunta sem resposta é tratada como “não”;
    • linhas 40–45: a resposta: fecha a janela e resolve a promise;
    • linha 50: o provider exibe a própria janela quando há uma pergunta.

// 16-confirmation/src/confirmation/ConfirmDialog.tsx
// -------------------------------------------------------------------------
// O diálogo modal:
//   - mesmas classes do Bootstrap de um modal de verdade (.modal, .modal-backdrop),
//     exibido "fixo no código" (show d-block): sem necessidade do
//     JavaScript do Bootstrap;
//   - createPortal: o diálogo é colocado no final de <body>, e não onde o
//     componente está escrito. Assim ele escapa dos estilos de seus
//     ancestrais (overflow, z-index...), continuando a ser um componente
//     React comum (props, eventos);
//   - teclado: Escape = Não; o foco é colocado no botão "Não" (a escolha
//     menos perigosa) ao abrir.
// =========================================================================

import { useEffect } from 'react';
import { createPortal } from 'react-dom';

interface Props {
  message: string;
  onAnswer: (yes: boolean) => void;
}

export default function ConfirmDialog({ message, onAnswer }: Props) {
  // Escape, em qualquer lugar da página: um listener colocado no DOCUMENT
  // ao abrir, removido ao fechar (limpeza do efeito)
  useEffect(() => {
    const onKeydown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onAnswer(false);
    };
    document.addEventListener('keydown', onKeydown);
    return () => document.removeEventListener('keydown', onKeydown);
  }, [onAnswer]);

  return createPortal(
    <>
      <div className="modal-backdrop fade show"></div>
      {/* um clique no fundo (e não no diálogo) significa "não" */}
      <div className="modal fade show d-block" tabIndex={-1} role="alertdialog" aria-modal="true"
           onClick={(e) => e.target === e.currentTarget && onAnswer(false)}>
        <div className="modal-dialog modal-dialog-centered">
          <div className="modal-content">
            <div className="modal-header">
              <h5 className="modal-title">Confirmation</h5>
              <button type="button" className="btn-close" aria-label="Close" onClick={() => onAnswer(false)}></button>
            </div>
            <div className="modal-body">
              <p className="mb-0">{message}</p>
            </div>
            <div className="modal-footer">
              {/* autoFocus: o React dá o foco a este botão quando ele aparece */}
              <button type="button" className="btn btn-secondary" autoFocus onClick={() => onAnswer(false)}>No, go back</button>
              <button type="button" className="btn btn-danger" onClick={() => onAnswer(true)}>Yes, confirm</button>
            </div>
          </div>
        </div>
      </div>
    </>,
    document.body,
  );
}
    • linhas 27–33: a tecla Esc, em qualquer lugar da página: um listener anexado a DOCUMENT quando a janela abre e removido quando ela fecha (limpeza). Essa é tipicamente a função de um efeito: conectar-se a algo externo ao [React];
    • linha 35: [createPortal(jsx, element)] exibe JSX em um lugar diferente de onde o componente é escrito: aqui, no final de [<body>]. A janela escapa assim dos estilos de seus ancestrais ([overflow], [z-index]...), continuando a ser um componente [React] comum (props, eventos);
